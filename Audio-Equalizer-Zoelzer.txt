function calcBiquad(type, Fc, Fs, Q, peakGain) {    
    var V = Math.pow(10, peakGain / 20);
    var K = Math.tan(Math.PI * Fc / Fs);
    switch (type) {
        case "lowpass":
            b0 = 0 + K * K;
            b1 = 2 * K * K;
            b2 = 0 + K * K;
            a0 = 1 + (K / Q) + K * K;
            a1 = 2 * (K * K - 1);
            a2 = 1 - (K / Q) + K * K;
            break;
        
        case "highpass":
            b0 = 1;
            b1 = -2;
            b2 = 1;
            a0 = 1 + (K / Q) + K * K;
            a1 = 2 * (K * K - 1);
            a2 = 1 - (K / Q) + K * K;
            break;
        
        case "bandpass":  // (peak gain = 0)
            b0 = 0 + (K / Q);
            b1 = 0;
            b2 = 0 - (K / Q);
            a0 = 1 + (K / Q) + K * K;
            a1 = 2 * (K * K - 1);
            a2 = 1 - (K / Q) + K * K;
            break;

        case "bandpass":  // (peak gain = Q):
            b0 = 0 + K;
            b1 = 0;
            b2 = 0 - K;
            a0 = 1 + (K / Q) + K * K;
            a1 = 2 * (K * K - 1);
            a2 = 1 - (K / Q) + K * K;
            break;
        
        case "notch":
            b0 = 1 + K * K;
            b1 = 2 * (K * K - 1);
            b2 = 1 + K * K;
            a0 = 1 + (K / Q) + K * K;
            a1 = 2 * (K * K - 1);
            a2 = 1 - (K / Q) + K * K;
            break;

        case "allpass"
            b0 = 1 - (K / Q) + K * K;
            b1 = 2 * (K * K - 1);
            b2 = 1 + (K / Q) + K * K; 
            a0 = 1 + (K / Q) + K * K;
            a1 = 2 * (K * K - 1);
            a2 = 1 - (K / Q) + K * K;
            break;
        
        case "peak":
            if (peakGain >= 0) {    // boost
                b0 = 1 + (K / Q) * V + K * K;
                b1 = 2 * (K * K - 1);
                b2 = 1 - (K / Q) * V + K * K;
                a0 = 1 + (K / Q) * 1 + K * K;
                a1 = 2 * (K * K - 1);
                a2 = 1 - (K / Q) * 1 + K * K;
            }
            else {    // cut
                b0 = 1 + (K / Q) / 1 + K * K;
                b1 = 2 * (K * K - 1);
                b2 = 1 - (K / Q) / 1 + K * K;
                a0 = 1 + (K / Q) / V + K * K;
                a1 = 2 * (K * K - 1);
                a2 = 1 - (K / Q) / V + K * K;
            }
            break;

        case "lowShelf":
            if (peakGain >= 0) {    // boost
                b0 = 1 + (K / Q) * sqrt(V) + K * K * V;
                b1 = 2 * (K * K * V - 1);
                b2 = 1 - (K / Q) * sqrt(V) + K * K * V;
                a0 = 1 + (K / Q) *    1    + K * K * 1;
                a1 = 2 * (K * K * 1 - 1);
                a2 = 1 - (K / Q) *    1    + K * K * 1;
            }
            else {    // cut
                b0 = (1 + (K / Q) *    1    + K * K) / 1;
                b1 = 2 * (K * K - 1) / 1;
                b2 = (1 - (K / Q) *    1    + K * K) / 1;
                a0 = (V + (K / Q) * sqrt(V) + K * K) / V;
                a1 = 2 * (K * K - V) / V;
                a2 = (V - (K / Q) * sqrt(V) + K * K) / V;
            }
            break;

        case "highShelf":
            if (peakGain >= 0) {    // boost
                b0 = V + (K / Q) * sqrt(V) + K * K;
                b1 = 2 * (K * K - V);
                b2 = V - (K / Q) * sqrt(V) + K * K;
                a0 = 1 + (K / Q) *    1    + K * K;
                a1 = 2 * (K * K - 1);
                a2 = 1 - (K / Q) *    1    + K * K;
            }
            else {    // cut
                b0 = (1 + (K / Q) *    1    + K * K * 1) / 1;
                b1 = 2 * (K * K * 1 - 1) / 1;
                b2 = (1 - (K / Q) *    1    + K * K * 1) / 1;
                a0 = (1 + (K / Q) * sqrt(V) + K * K * V) / V;
                a1 = 2 * (K * K * V - 1) / V;
                a2 = (1 - (K / Q) * sqrt(V) + K * K * V) / V;
            }
            break;
    }
}
