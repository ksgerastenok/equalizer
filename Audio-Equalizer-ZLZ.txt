function calcBiquad(type, Fc, Fs, Q, gain) {    
    var V = pow(10, gain / 20);
    var K = tan(PI * Fc / Fs);

    switch (type) {
        case "lowpass": {
            b0 = 0 + K * K;
            b1 = 2 * (K * K - 0);
            b2 = 0 + K * K;
            a0 = 1 + (K / Q) + K * K;
            a1 = 2 * (K * K - 1);
            a2 = 1 - (K / Q) + K * K;
        }

        case "highpass": {
            b0 = 1;
            b1 = -2;
            b2 = 1;
            a0 = 1 + (K / Q) + K * K;
            a1 = 2 * (K * K - 1);
            a2 = 1 - (K / Q) + K * K;
        }

        case "bandpass": {
            b0 = 0 + (K / Q);
            b1 = 0;
            b2 = 0 - (K / Q);
            a0 = 1 + (K / Q) + K * K;
            a1 = 2 * (K * K - 1);
            a2 = 1 - (K / Q) + K * K;
        }

        case "bandpass": {
            b0 = 0 + K;
            b1 = 0;
            b2 = 0 - K;
            a0 = 1 + (K / Q) + K * K;
            a1 = 2 * (K * K - 1);
            a2 = 1 - (K / Q) + K * K;
        }

        case "notch": {
            b0 = 1 + K * K;
            b1 = 2 * (K * K - 1);
            b2 = 1 + K * K;
            a0 = 1 + (K / Q) + K * K;
            a1 = 2 * (K * K - 1);
            a2 = 1 - (K / Q) + K * K;
        }

        case "allpass": {
            b0 = 1 - (K / Q) + K * K;
            b1 = 2 * (K * K - 1);
            b2 = 1 + (K / Q) + K * K; 
            a0 = 1 + (K / Q) + K * K;
            a1 = 2 * (K * K - 1);
            a2 = 1 - (K / Q) + K * K;
        }

        case "peak": {
            if (gain >= 0) {
                b0 = 1 + (K / Q) * V + K * K;
                b1 = 2 * (K * K - 1);
                b2 = 1 - (K / Q) * V + K * K;
                a0 = 1 + (K / Q) * 1 + K * K;
                a1 = 2 * (K * K - 1);
                a2 = 1 - (K / Q) * 1 + K * K;
            }
            else {
                b0 = 1 + (K / Q) / 1 + K * K;
                b1 = 2 * (K * K - 1);
                b2 = 1 - (K / Q) / 1 + K * K;
                a0 = 1 + (K / Q) / V + K * K;
                a1 = 2 * (K * K - 1);
                a2 = 1 - (K / Q) / V + K * K;
            }
        }

        case "lowShelf": {
            if (gain >= 0) {
                b0 =(1 + (K / Q) /    1    + K * K / 1);
                b1 = 2 * (K * K / 1 - 1);
                b2 =(1 - (K / Q) /    1    + K * K / 1);
                a0 =(1 + (K / Q) / sqrt(V) + K * K / V);
                a1 = 2 * (K * K / V - 1);
                a2 =(1 - (K / Q) / sqrt(V) + K * K / V);
            }
            else {
                b0 =(1 + (K / Q) * sqrt(V) + K * K * V);
                b1 = 2 * (K * K * V - 1);
                b2 =(1 - (K / Q) * sqrt(V) + K * K * V);
                a0 =(1 + (K / Q) *    1    + K * K * 1);
                a1 = 2 * (K * K * 1 - 1);
                a2 =(1 - (K / Q) *    1    + K * K * 1);
            }
        }

        case "highShelf": {
            if (gain >= 0) {
                b0 =(1 + (K / Q) / sqrt(V) + K * K / V) * V;
                b1 = 2 * (K * K / V - 1) * V;
                b2 =(1 - (K / Q) / sqrt(V) + K * K / V) * V;
                a0 =(1 + (K / Q) /    1    + K * K / 1) * 1;
                a1 = 2 * (K * K / 1 - 1) * 1;
                a2 =(1 - (K / Q) /    1    + K * K / 1) * 1;
            }
            else {
                b0 =(1 + (K / Q) *    1    + K * K * 1) / 1;
                b1 = 2 * (K * K * 1 - 1) / 1;
                b2 =(1 - (K / Q) *    1    + K * K * 1) / 1;
                a0 =(1 + (K / Q) * sqrt(V) + K * K * V) / V;
                a1 = 2 * (K * K * V - 1) / V;
                a2 =(1 - (K / Q) * sqrt(V) + K * K * V) / V;
            }
        }
    }
}
