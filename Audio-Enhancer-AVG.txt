---------------------------------------------------------
w = 0.0 => No action
mono   = (L + R) / 2.0
stereo = (L - R) / 2.0 * (w + 1.0)
L = mono + stereo
R = mono - stereo
---------------------------------------------------------
s = (L + R) / 2.0
L = L + w * (L - s) = s - (w + 1.0) * (s - L)
R = R + w * (R - s) = s - (w + 1.0) * (s - R)
---------------------------------------------------------
s = Avg(c)
c[i] = c[i] + w * (c[i] - s) = s - (w + 1.0) * (s - c[i])
---------------------------------------------------------
